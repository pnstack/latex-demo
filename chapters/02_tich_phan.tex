\section{Tích phân}

\subsection{Nguyên hàm}
$F$ là nguyên hàm của $f$ trên khoảng $K$ nếu $F'(x) = f(x)$ với mọi $x \in K$:
\[
\int f(x)\,dx = F(x) + C.
\]

\subsection{Công thức Newton--Leibniz}
\[
\int_a^b f(x)\,dx = F(b) - F(a).
\]

\subsection{Một số nguyên hàm thường gặp}
\begin{align*}
\int x^n\,dx &= \frac{x^{n+1}}{n+1} + C, \quad n \neq -1 \\
\int \frac{1}{x}\,dx &= \ln|x| + C \\
\int e^x\,dx &= e^x + C \\
\int \sin x\,dx &= -\cos x + C
\end{align*}
